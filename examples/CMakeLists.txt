function(add_vio_example name)
    add_executable(${name} ${name}.cpp)
    CmDepTargetLinkLibrary(${name} PRIVATE vio BoringSSL::SSL BoringSSL::Crypto)
    target_include_directories(${name} PRIVATE ${CMAKE_SOURCE_DIR}/src)
endfunction()
